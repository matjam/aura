# Media queries against a 1024x768 screen (the default device) unless the
# options say otherwise.

=== media types
--- media
all => true
screen => true
print => false
only screen => true
not print => true
not screen => false
SCREEN => true
tv => false
speech => false
all and (color) => true
screen, print => true
print, tv => false

=== width and height
--- media

(min-width: 1025px) => false
(max-width: 1024px) => true
(width: 1024px) => true
(width) => true
(height: 768px) => true
(min-width: 64em) => true
(min-width: 65em) => false
(max-width: 100vw) => true
(min-width: calc(500px + 500px)) => true
(min-width: 1000) => false
(min-width: 0) => true

screen and (max-width: 600px) => false

=== range syntax
--- media
(width > 1000px) => true
(width >= 1024px) => true
(width > 1024px) => false
(1000px < width) => true
(500px < width < 2000px) => true
(500px < width < 800px) => false
(2000px > width > 500px) => true
(400px <= height <= 768px) => true
(width < 500px < 600px) => false

=== boolean logic
--- media
(min-width: 100px) and (max-width: 200px) => false
not all and (max-width: 200px) => true
(not (max-width: 200px)) => true
((min-width: 100px) or (max-width: 50px)) => true
((min-width: 2000px) or (max-width: 50px)) => false
# Media Queries 4 allows or at the top level of a condition
(min-width: 100px) or (max-width: 50px) => true
screen and (min-width: 100px) or (max-width: 50px) => false
(min-width: 100px) and (max-width: 50px) or (color) => false
not (color) => false
